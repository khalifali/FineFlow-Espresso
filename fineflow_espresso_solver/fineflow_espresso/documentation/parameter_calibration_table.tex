{\footnotesize
\textbf{Purpose.} Compact reference for the numerical inputs in \texttt{ModelParameters}. The table separates directly measured inputs, CT-derived structure, hydraulic experiments, mass-in-cup measurements, CFD--DEM support, and numerical choices. The general porosity baseline is $\varepsilon_0=0.55$ when \texttt{initial\_porosity\_mode=given}. For \texttt{mass\_height\_density}, both \texttt{coffee\_mass\_kg} and \texttt{particle\_density\_kg\_m3} are mandatory and
\[
\varepsilon_0=1-\frac{m_{\rm coffee}}{\rho_p A H}.
\]
}

\vspace{2mm}
{\footnotesize
\textbf{CFD--DEM case map:}
Case 01 = clean reference / $K_0$; Case 02 = held-fines blockage / $K/K_0(m_d)$; Case 03 = dilute capture / $k_{\rm dep}$ and $D_{\rm ax}$; Case 04 = progressive loading / capacity and blockage evolution; Case 05 = remobilization / $k_{\rm det}$, $G_{\rm crit}$, exponent and escape.

\textbf{Categorical switches:}
\texttt{initial\_porosity\_mode} = \texttt{given} or \texttt{mass\_height\_density};
\texttt{permeability\_model} = \texttt{exponential}, \texttt{kozeny\_carman}, or tabulated legacy \texttt{pnm\_table};
\texttt{hydraulic\_model} = \texttt{darcy} or \texttt{darcy\_forchheimer};
\texttt{washout\_escape\_model} = \texttt{unity}, \texttt{logistic}, or tabulated legacy \texttt{pnm\_table};
\texttt{control\_mode} = \texttt{constant}, \texttt{pi}, \texttt{pressure}, or \texttt{flow}.
}
